    88→  /* ── Step 1: submit order ─────────────────────────────────────────────── */
    89→
    90→  async function submitOrder(e: React.FormEvent) {
    91→    e.preventDefault();
    92→    setServerError(null);
    93→    setFieldErrs({});
    94→    setSubmitting(true);
    95→    try {
    96→      const res = await fetch("/api/order", {
    97→        method: "POST",
    98→        headers: { "Content-Type": "application/json" },
    99→        body: JSON.stringify({
   100→          customer: {
   101→            fullName: form.fullName,
   102→            mobile: form.mobile,
   103→            email: form.email,
   104→            address: form.address,
   105→            city: form.city,
   106→            state: form.state,
   107→            pincode: form.pincode,
   108→          },
   109→          items: [{ productId, quantity: qty }],
   110→          note: form.note,
   111→          company: form.company,
   112→        }),
   113→      });
   114→      const data = await res.json();
   115→
   116→      if (res.status === 429) {
   117→        setServerError(
   118→          "You're ordering a little too quickly. Please wait a moment and try again.",
   119→        );
   120→        return;
   121→      }
   122→      if (res.status === 400 && data?.errors) {
   123→        setFieldErrs(data.errors);
   124→        setServerError("Please review the highlighted fields below.");
   125→        return;
   126→      }
   127→      if (!res.ok) {
   128→        setServerError(
   129→          "We couldn't submit your order right now. Please try again in a moment or contact us on WhatsApp.",
   130→        );
   131→        return;
   132→      }
   133→
   134→      setConfirmed({
   135→        orderId: data.orderId,
   136→        deliveryLabel: data.deliveryLabel,
   137→        total: data.total,
   138→        items: data.items,
   139→      });
   140→      setStep("payment");
   141→      toast({
   142→        title: "Order received",
   143→        description: `Your order ID is ${data.orderId}. A confirmation email is on its way.`,
   144→      });
   145→    } catch {
   146→      setServerError(
   147→        "We couldn't reach the server. Please check your connection and try again, or contact us on WhatsApp.",
   148→      );
   149→    } finally {
   150→      setSubmitting(false);
   151→    }
   152→  }
   153→
   154→  /* ── Step 2: payment ─────────────────────────────────────────────────── */

   156→  async function copyUpi() {
   157→    try {
   158→      await navigator.clipboard.writeText(upiId);

   162→      toast({ title: "Couldn't copy", description: `UPI ID: ${upiId}` });
   163→    }
   164→  }
   165→
   166→  async function submitUtr(): Promise<boolean> {
   167→    const trimmed = utr.trim();
   168→    if (!trimmed) return true; // optional
   169→    if (trimmed.length < 6) {
   170→      setFieldErrs({ utr: "Reference looks too short." });
   171→      return false;
   172→    }
   173→    setUtrSubmitting(true);
   174→    try {
   175→      const res = await fetch("/api/order/payment-note", {
   176→        method: "POST",
   177→        headers: { "Content-Type": "application/json" },
   178→        body: JSON.stringify({ orderId: confirmed?.orderId, utr: trimmed }),
   179→      });
   180→      if (res.ok) {
   181→        toast({
   182→          title: "Payment reference received",
   183→          description: "We'll match it with your payment shortly.",
   184→        });
   185→        return true;
   186→      }
   187→      // Non-blocking: UTR is optional metadata; confirmation continues either way.
   188→      toast({
   189→        title: "Reference couldn't be submitted",
   190→        description: "No problem — share it with us on WhatsApp instead.",
   191→      });
   192→      return true;
   193→    } catch {
   194→      return true; // optional field, never block confirmation
   195→    } finally {
   196→      setUtrSubmitting(false);
   197→    }
   198→  }
   199→
